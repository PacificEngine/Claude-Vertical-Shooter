# Power-ups Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add shootable crates that drop pickups, and six timed power-ups (Tank, Spread, Shield, Rapid, Pierce, Bomb) the squad collects by walking into them.

**Architecture:** One active-power-up slot (`state.power`) that existing rules consult, plus three new rules (`powerTimer`, `bulletsVsCrates`, `pickupsVsSquad`). `step` stays a flat pipeline of pure rules; the renderer stays a pure function of state.

**Tech Stack:** TypeScript (strict), Vite, Vitest, Playwright, ESLint. Package manager: yarn.

**Spec:** `docs/superpowers/specs/2026-10-21-power-ups-design.md` (builds on `docs/superpowers/specs/2026-10-21-core-engine-design.md`)

## Global Constraints

- Package manager is **yarn only**. Never npm, pnpm or bun.
- `src/core/` has no DOM, no canvas and no `Date.now()`. Rules are pure and return new state; inputs are never mutated.
- Randomness comes only from the seeded RNG in `state.rng`. Zombie ids come from `state.nextId`.
- New `step` order: move, spawn, powerTimer, fire, bulletsVsGates, bulletsVsZombies, bulletsVsCrates, squadVsGates, pickupsVsSquad, zombiesVsSquad, end.
- Exactly one power-up is active at a time (`state.power`); collecting a non-bomb one replaces it. Bomb is instant and never stored in `power`.
- In the renderer, `arc` is used only for soldiers (tests count arcs).
- `@playwright/test` stays pinned at exactly `1.63.0` and `typescript` at `^6`. Never run `playwright install` or download a browser (a matching Chromium is cached).
- TDD always: failing test, see it fail, minimal code, see it pass, commit. Never refactor and change behavior in the same commit.
- Work on branch `feature/power-ups`. Never commit to `main`. Commit messages explain *why*. No Claude signature on commits.
- No Snyk scan in this repo.

```ts
import type { GameState } from "../types";

export function powerTimer(state: GameState, dt: number): GameState {
  if (!state.power) return state;
  const remaining = state.power.remaining - dt;
  return { ...state, power: remaining > 0 ? { ...state.power, remaining } : null };
}
```

```ts
import { CRATE_SIZE } from "../constants";
import type { Bullet, GameState } from "../types";

export function bulletsVsCrates(state: GameState): GameState {
  const crates = state.crates.map((c) => ({ ...c }));
  const bullets: Bullet[] = [];

  for (const bullet of state.bullets) {
    const target = crates.find(
      (c) =>
        c.hp > 0 && Math.abs(bullet.x - c.x) <= CRATE_SIZE / 2 && Math.abs(bullet.y - c.y) <= CRATE_SIZE / 2,
    );
    if (target) target.hp -= bullet.damage;
    else bullets.push(bullet);
  }

  const released = crates.filter((c) => c.hp <= 0).map((c) => ({ x: c.x, y: c.y, power: c.power }));
  return {
    ...state,
    bullets,
    crates: crates.filter((c) => c.hp > 0),
    pickups: [...state.pickups, ...released],
  };
}
```

```ts
import { BOMB_DAMAGE, PICKUP_SIZE, POWER_DURATION, SHIELD_CHARGES, SQUAD_HALF_WIDTH, SQUAD_Y } from "../constants";
import type { GameState, Pickup } from "../types";

const touches = (squadX: number, pickup: Pickup): boolean =>
  Math.abs(squadX - pickup.x) <= SQUAD_HALF_WIDTH + PICKUP_SIZE / 2 &&
  Math.abs(pickup.y - SQUAD_Y) <= PICKUP_SIZE;

export function pickupsVsSquad(state: GameState): GameState {
  let power = state.power;
  let zombies = state.zombies;
  const pickups: Pickup[] = [];

  for (const pickup of state.pickups) {
    if (!touches(state.squad.x, pickup)) {
      pickups.push(pickup);
    } else if (pickup.power === "bomb") {
      zombies = zombies
        .map((z) => (z.y > 0 ? { ...z, hp: z.hp - BOMB_DAMAGE } : z))
        .filter((z) => z.hp > 0);
    } else {
      power = {
        kind: pickup.power,
        remaining: POWER_DURATION[pickup.power],
        charges: pickup.power === "shield" ? SHIELD_CHARGES : 0,
      };
    }
  }

  return { ...state, pickups, zombies, power };
}
```

```ts
import {
  BASE_DAMAGE, FIRE_INTERVAL, MAX_BULLETS, MAX_VISIBLE, RAPID_RATE_MULT, SPREAD_VX, SQUAD_Y,
  TANK_DAMAGE_MULT, TANK_RATE_MULT, TANK_SPLASH,
} from "../constants";
import { soldierOffsets } from "../formation";
import type { ActivePowerKind, Bullet, GameState } from "../types";

const bullet = (x: number, y: number, damage: number, extra: Partial<Bullet> = {}): Bullet => ({
  x, y, damage, passedGates: [], vx: 0, pierce: false, hit: [], splash: 0, ...extra,
});

function intervalFor(kind: ActivePowerKind | undefined): number {
  if (kind === "rapid") return FIRE_INTERVAL * RAPID_RATE_MULT;
  if (kind === "tank") return FIRE_INTERVAL * TANK_RATE_MULT;
  return FIRE_INTERVAL;
}

function soldierBullets(state: GameState, visible: number, kind: ActivePowerKind | undefined): Bullet[] {
  const damage = (BASE_DAMAGE * state.squad.count) / visible;
  const speeds = kind === "spread" ? [-SPREAD_VX, 0, SPREAD_VX] : [0];
  return soldierOffsets(visible).flatMap(({ dx, dy }) =>
    speeds.map((vx) => bullet(state.squad.x + dx, SQUAD_Y + dy, damage, { vx, pierce: kind === "pierce" })),
  );
}

const tankShell = (state: GameState): Bullet[] => [
  bullet(state.squad.x, SQUAD_Y, BASE_DAMAGE * state.squad.count * TANK_DAMAGE_MULT, { splash: TANK_SPLASH }),
];

export function fireRule(state: GameState, dt: number): GameState {
  const timer = state.fireTimer - dt;
  if (timer > 0) return { ...state, fireTimer: timer };

  const kind = state.power?.kind;
  const interval = intervalFor(kind);
  const visible = Math.min(state.squad.count, MAX_VISIBLE);
  if (visible === 0) return { ...state, fireTimer: interval };

  const room = Math.max(0, MAX_BULLETS - state.bullets.length);
  const fired = (kind === "tank" ? tankShell(state) : soldierBullets(state, visible, kind)).slice(0, room);

  return { ...state, bullets: [...state.bullets, ...fired], fireTimer: timer + interval };
}
```

```ts
import { ZOMBIE_RADIUS } from "../constants";
import type { Bullet, GameState, Zombie } from "../types";

const overlaps = (bullet: Bullet, z: Zombie): boolean =>
  z.hp > 0 && Math.abs(bullet.x - z.x) <= ZOMBIE_RADIUS && Math.abs(bullet.y - z.y) <= ZOMBIE_RADIUS;

export function bulletsVsZombies(state: GameState): GameState {
  const zombies = state.zombies.map((z) => ({ ...z }));
  const bullets: Bullet[] = [];

  for (const bullet of state.bullets) {
    if (bullet.pierce) {
      const hit = [...bullet.hit];
      for (const z of zombies) {
        if (overlaps(bullet, z) && !hit.includes(z.id)) {
          z.hp -= bullet.damage;
          hit.push(z.id);
        }
      }
      bullets.push({ ...bullet, hit });
      continue;
    }

    const target = zombies.find((z) => overlaps(bullet, z));
    if (!target) {
      bullets.push(bullet);
    } else if (bullet.splash > 0) {
      for (const z of zombies) {
        if (z.hp > 0 && Math.hypot(z.x - bullet.x, z.y - bullet.y) <= bullet.splash) z.hp -= bullet.damage;
      }
    } else {
      target.hp -= bullet.damage;
    }
  }

  return { ...state, bullets, zombies: zombies.filter((z) => z.hp > 0) };
}
```
